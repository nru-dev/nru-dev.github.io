Обо мне

Меня зовут Silvi. Занимаюсь реверсом игровых клиентов и системной разработкой под Windows. Мне интересно то, что обычно остаётся за кадром: структуры, вызовные соглашения, графические хуки, память процесса.

veyr d3d9 · x86 · EndScene

Перехват EndScene без угадывания адресов

Временный хук на CreateDevice, ловим живой vtable устройства и ставим прямые entry-hooks. Оригинал уезжает в трэмплин, teardown снимает Reset раньше EndScene. 

// u32 stdcall IDirect3DDevice9::EndScene(IDirect3DDevice9* this)
static HRESULT __stdcall hooked_end_scene(void* device) {
    auto* rt = runtime::instance();
    if (rt && rt->frame_ready(device)) {
        rt->render_overlay(device);   // player circle / diagnostics
    }
    return g_original_end_scene(device);  // trampoline
}

// install only after capture hooks die
bool install_renderer_hooks(void* device) {
    auto vtbl = *(void***)device;
    auto end_scene = vtbl[42];          // IDirect3DDevice9::EndScene
    auto reset     = vtbl[16];          // IDirect3DDevice9::Reset
    return hooks::commit({
        {end_scene, &hooked_end_scene, &g_original_end_scene},
        {reset,     &hooked_reset,     &g_original_reset},
    }, /*fail_closed=*/ true);
}
veyr inject · loader · PE32

Инъекция до первого кадра клиента

Wow.exe стартует приостановленным, loader поднимает нити после bootstrap, инжектит veyr.dll и взводит capture-хуки до создания D3D-устройства. 

// Wow.exe suspended → inject veyr.dll → resume
int launch_client(const wchar_t* wow_path) {
    STARTUPINFOW si{sizeof(si)};
    PROCESS_INFORMATION pi{};
    CreateProcessW(wow_path, nullptr, nullptr, nullptr,
                   FALSE, CREATE_SUSPENDED, nullptr, nullptr, &si, &pi);

    auto remote = inject::allocate(pi.hProcess, kDllPath);
    inject::write_image(pi.hProcess, remote, kDllPath);
    inject::thread_start(pi.hProcess, remote.load_fn, remote.base);

    // capture hooks armed inside DllMain; resume only after
    // Direct3DCreate9 trampoline is live - fail-closed
    if (!capture::wait_armed(pi.hProcess, 2000)) {
        capture::abort_suspend(pi.hProcess, pi.hThread);
        return 1;
    }
    ResumeThread(pi.hThread);
    return 0;
}
veyr offsets · CGCamera · build 12340

Восстановление world-to-screen

Камера собирается из живого CGCamera: базис, FOV и origin читаются через валидированные смещения. Никаких «средних значений» - только проверенные поля. 

// recovered field map · build 12340 (x86)
struct CGCamera {              // size probe: 0x1C0+
    void*   vtable;        // +0x00  vtable checks only
    C3Vector origin;        // +0x0C  world position
    C3Vector right;         // +0x18  basis vectors
    C3Vector up;
    C3Vector forward;       // +0x30  look direction
    float   fov_y;         // +0x54  validated vs live device
};

bool world_to_screen(const CGCamera& cam, const C3Vector& world,
                    int w, int h, C2Vector* out) {
    auto rel = world - cam.origin;
    auto x = rel.dot(cam.right);
    auto y = rel.dot(cam.up);
    auto z = rel.dot(cam.forward);
    if (z <= 0.01f) return false;  // behind camera
    auto f = 1.0f / tanf(cam.fov_y * 0.5f);
    out->x = (x * f / z + 1.0f) * w * 0.5f;
    out->y = (1.0f - y * f / z * (w / (float)h)) * h * 0.5f;
    return true;
}
veyr ABI · terrain · fail-closed

CGWorldFrame::Intersect - рейк по ландшафту

Вызовный контракт восстановлен из бинаря, но path включается только после живой in-game пробы. Без неё - безопасный статический круг. 

// native contract recovered from build 12340
using IntersectFn = bool (__thiscall*)(
    void* world_frame,
    const C3Vector& from,
    const C3Vector& to,
    uint32_t flags,
    HitRecord* out);

bool ray_terrain(void* world_frame, const C3Vector& a,
               const C3Vector& b, HitRecord* hit) {
    if (!probe::collision_profile_valid()) {
        return false;  // fail-closed · static ring only
    }
    auto fn = reinterpret_cast<IntersectFn>(kIntersect);
    HitRecord tmp{};
    if (!fn(world_frame, a, b, kTerrainFlag, &tmp)) return false;
    if (!validate_hit_record(tmp))      return false;
    *hit = tmp;
    return true;
}
veyr probe · vtable · mapped modules

Проверка оффсета перед доверием

Смещение - не факт, пока не сошлись mapped-module, vtable-signature и живой указатель. Иначе адрес отбрасывается и путь не используется. 

enum class Probe { Unmapped, BadVtable, Untrusted, Trusted };

Probe probe_gx_device(uintptr_t module_base, uintptr_t offset) {
    auto addr = module_base + offset;
    if (!is_mapped_readable(addr, sizeof(void*)))
        return Probe::Unmapped;

    auto object = *reinterpret_cast<void**>(addr);
    if (!is_mapped_readable(object, 32 * sizeof(void*)))
        return Probe::BadVtable;

    auto vtbl = *reinterpret_cast<void***>(object);
    if (!looks_like_d3d9_device_vtable(vtbl))
        return Probe::BadVtable;

    // legacy GxDevice offset is never source of truth for hooks
    return capture::live_device() == object
        ? Probe::Trusted
        : Probe::Untrusted;
}
